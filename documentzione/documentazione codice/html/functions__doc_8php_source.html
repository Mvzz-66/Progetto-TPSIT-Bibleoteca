53function cercaUtente($smartCard, $password) {
54 $percorso = __DIR__ . '/utenti.txt';
55 if (!file_exists($percorso)) return null;
56
57 $file = fopen($percorso, 'r');
58 if (!$file) return null;
59
60 while (($line = fgets($file)) !== false) {
61 $cols = explode(';', trim($line));
62 if (count($cols) < 5) continue;
63
64 // password_verify() confronta la password in chiaro con l'hash salvato nel file
65 if (trim($cols[0]) === $smartCard && password_verify($password, trim($cols[4]))) {
66 fclose($file);
67 return ['nome' => trim($cols[1]), 'cognome' => trim($cols[2])];
68 }
69 }
70 fclose($file);
71 return null;
72}

90function generaSmartCard() {
91 $percorso = __DIR__ . '/utenti.txt';
92 do {
93 $codice = 'SC-' . strtoupper(substr(str_shuffle('ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789'), 0, 8));
94 $codiceEsiste = false;
95 if (file_exists($percorso)) {
96 foreach (file($percorso) as $riga) {
97 $cols = explode(';', trim($riga));
98 if (count($cols) >= 1 && trim($cols[0]) === $codice) {
99 $codiceEsiste = true;
100 break;
101 }
102 }
103 }
104 } while ($codiceEsiste);
105 return $codice;
106}

151function autocompleteAjax() {
152 $q = strtolower(trim($_GET['q'] ?? ''));
153 $results = [];
154
155 if (strlen($q) >= 2) {
156 $file = fopen(__DIR__ . '/lista-libri.txt', 'r');
157 if ($file) {
158 fgets($file); // salta intestazione
159 while (($line = fgets($file)) !== false && count($results) < 8) {
160 $cols = explode(';', trim($line));
161 if (count($cols) < 4) continue;
162 $titolo = $cols[2];
163 $autore = $cols[3];
164 $isbn = $cols[7] ?? '';
165 if (
166 str_contains(strtolower($titolo), $q) ||
167 str_contains(strtolower($autore), $q) ||
168 str_contains(strtolower($isbn), $q)
169 ) {
170 $results[] = ['titolo' => trim($titolo), 'autore' => trim($autore)];
171 }
172 }
173 fclose($file);
174 }
175 }
176
177 header('Content-Type: application/json');
178 echo json_encode($results);
179 exit();
180}

324function aggiornaDisponibilita($idLibro, $nuovoVal, $dataInizio = null, $dataFine = null) {
325 $percorso = __DIR__ . '/lista-libri.txt';
326 $righe = file($percorso);
327 if ($righe === false) return false;
328 $trovato = false;
329 foreach ($righe as $i => $riga) {
330 $cols = explode(';', trim($riga));
331 if (count($cols) < 11) continue;
332 if (trim($cols[1]) === $idLibro) {
333 $cols[10] = $nuovoVal;
334 if ($dataInizio !== null) $cols[11] = $dataInizio;
335 if ($dataFine !== null) $cols[12] = $dataFine;
336 $righe[$i] = implode(';', $cols) . "\n";
337 $trovato = true;
338 break;
339 }
340 }
341 if (!$trovato) return false;
342 return file_put_contents($percorso, implode('', $righe)) !== false;
343}
